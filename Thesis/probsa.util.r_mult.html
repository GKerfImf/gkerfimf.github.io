Library probsa.util.r_mult

Require Export Reals.
From mathcomp Require Import ssreflect ssrbool ssrfun eqtype choice fintype bigop seq.
Import Monoid.

From discprob.prob Require Export prob countable.

Lemma Rmult_eq_compat :
  ∀ r1 r2 r3 r4 : R, (r1 = r2 → r3 = r4 → r1 × r3 = r2 × r4)%R.
Proof. by intros; subst. Qed.

Lemma Rmult_associative: associative Rmult.
Proof. intros ?; apply: Hierarchy.mult_assoc. Qed.
Lemma Rmult_left_id: left_id R1 Rmult.
Proof. intros ?; apply: Rmult_1_l. Qed.
Lemma Rmult_right_id: right_id R1 Rmult.
Proof. intros ?; apply: Rmult_1_r. Qed.
Lemma Rmult_commutative: commutative Rmult.
Proof. intros ?; apply: Rmult_comm. Qed.

Canonical Rmult_monoid := Law Rmult_associative Rmult_left_id Rmult_right_id.
Canonical Rmult_comoid := ComLaw Rmult_commutative.
