Section Pair.

  Context {A B: Type}.
  Variable p: A × B.
  Definition pair_1st := fst p.
  Definition pair_2nd := snd p.

End Pair.

Section Triple.

  Context {A B C: Type}.
  Variable p: A × B × C.
  Definition triple_1st (p: A × B × C) := fst (fst p).
  Definition triple_2nd := snd (fst p).
  Definition triple_3rd := snd p.

End Triple.

Definition make_sequence {T: Type} (opt: option T) :=
  match opt with
    | Some j ⇒ [:: j]
    | None ⇒ [::]
  end.
